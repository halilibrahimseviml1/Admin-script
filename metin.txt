local espEnabled = false
local espSize = 1.0
local espShowHealth = true
local espConnections = {}
local healthConnections = {}

local function removeESP(player)
    if player.Character and player.Character:FindFirstChild("TeamESP") then
        player.Character.TeamESP:Destroy()
    end
    if healthConnections[player] then
        healthConnections[player]:Disconnect()
        healthConnections[player] = nil
    end
end

local function createESP(player)
    if player == localPlayer then return end

    local function addVisuals(character)
        if not character:WaitForChild("HumanoidRootPart", 5) then return end
        local head = character:WaitForChild("Head", 5)
        if not head then return end

        if character:FindFirstChild("TeamESP") then
            character.TeamESP:Destroy()
        end

        local folder = Instance.new("Folder")
        folder.Name = "TeamESP"
        folder.Parent = character

        local highlight = Instance.new("Highlight")
        highlight.Name = "BoxESP"
        highlight.Adornee = character
        highlight.FillTransparency = 0.5
        highlight.OutlineTransparency = 0
        highlight.Enabled = espEnabled
        highlight.Parent = folder

        local billboard = Instance.new("BillboardGui")
        billboard.Name = "NameESP"
        billboard.Adornee = head
        billboard.Size = UDim2.new(0, 80 * espSize, 0, 30 * espSize)
        billboard.StudsOffset = Vector3.new(0, 1.8, 0)
        billboard.AlwaysOnTop = true
        billboard.Enabled = espEnabled
        billboard.Parent = folder

        local textLabel = Instance.new("TextLabel")
        textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.BackgroundTransparency = 1
        textLabel.TextSize = 9 * espSize
        textLabel.Font = Enum.Font.SourceSansBold
        textLabel.TextWrapped = true
        textLabel.RichText = true
        textLabel.Parent = billboard

        local humanoid = character:WaitForChild("Humanoid", 5)

        local function updateVisuals()
            if not folder.Parent then return end

            local fillColor = Color3.fromRGB(255, 255, 0)
            local nameColor = Color3.fromRGB(255, 255, 0)

            if player.Team and localPlayer.Team then
                if player.Team == localPlayer.Team then
                    fillColor = Color3.fromRGB(0, 255, 0)
                    nameColor = Color3.fromRGB(0, 255, 0)
                else
                    fillColor = Color3.fromRGB(255, 0, 0)
                    nameColor = Color3.fromRGB(255, 0, 0)
                end
            end

            highlight.FillColor = fillColor

            local r, g, b = nameColor.R*255, nameColor.G*255, nameColor.B*255
            local coloredName = string.format('<font color="rgb(%d,%d,%d)">%s</font>', r, g, b, player.Name)

            local teamName = player.Team and player.Team.Name or ""
            local teamPart = ""
            if teamName ~= "" then
                teamPart = string.format(' <font color="rgb(255,255,255)">(%s)</font>', teamName)
            end

            local healthPart = ""
            if humanoid and espShowHealth then
                local hp = math.floor(humanoid.Health)
                local maxHp = math.floor(humanoid.MaxHealth)
                healthPart = string.format('\n<font color="rgb(255,255,0)">[%d/%d]</font>', hp, maxHp)
            end

            textLabel.Text = coloredName .. teamPart .. healthPart
        end

        updateVisuals()

        if humanoid then
            healthConnections[player] = humanoid.HealthChanged:Connect(updateVisuals)
        end

        player:GetPropertyChangedSignal("Team"):Connect(updateVisuals)
        localPlayer:GetPropertyChangedSignal("Team"):Connect(updateVisuals)
    end

    local conn = player.CharacterAdded:Connect(addVisuals)
    espConnections[player] = conn

    if player.Character then
        task.spawn(function()
            addVisuals(player.Character)
        end)
    end
end

local function refreshAllESP()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= localPlayer and plr.Character then
            local folder = plr.Character:FindFirstChild("TeamESP")
            if folder then
                for _, child in ipairs(folder:GetChildren()) do
                    if child:IsA("Highlight") or child:IsA("BillboardGui") then
                        child.Enabled = espEnabled
                    end
                    if child:IsA("BillboardGui") then
                        child.Size = UDim2.new(0, 80 * espSize, 0, 30 * espSize)
                        local label = child:FindFirstChild("TextLabel")
                        if label then
                            label.TextSize = 9 * espSize
                        end
                    end
                end
            end
        end
    end
end

local function initESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then
            createESP(player)
        end
    end
    refreshAllESP()
end

Players.PlayerAdded:Connect(function(player)
    if player ~= localPlayer then
        createESP(player)
    end
end)

Players.PlayerRemoving:Connect(function(player)
    if espConnections[player] then
        espConnections[player]:Disconnect()
        espConnections[player] = nil
    end
    removeESP(player)
end)

local function splitString(str, sep)
    local parts = {}
    for match in string.gmatch(str, "([^" .. sep .. "]+)") do
        table.insert(parts, match)
    end
    return parts
end
