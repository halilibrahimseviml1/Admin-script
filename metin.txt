local espEnabled = false
local espSize = 1.0
local espShowHealth = true
local espConnections = {}
local healthConnections = {}
local espCustomColor = Color3.fromRGB(255, 255, 0)
local espTeamColorMode = true

local function removeESP(player)
    if player.Character and player.Character:FindFirstChild("TeamESP") then
        player.Character.TeamESP:Destroy()
    end
    if healthConnections[player] then
        healthConnections[player]:Disconnect()
        healthConnections[player] = nil
    end
end

local function createESP(player)
    if player == localPlayer then return end

    local function addVisuals(character)
        if not character:WaitForChild("HumanoidRootPart", 5) then return end
        local head = character:WaitForChild("Head", 5)
        if not head then return end

        if character:FindFirstChild("TeamESP") then
            character.TeamESP:Destroy()
        end

        local folder = Instance.new("Folder")
        folder.Name = "TeamESP"
        folder.Parent = character

        local highlight = Instance.new("Highlight")
        highlight.Name = "BoxESP"
        highlight.Adornee = character
        highlight.FillTransparency = 0.5
        highlight.OutlineTransparency = 0
        highlight.Enabled = espEnabled
        highlight.Parent = folder

        local billboard = Instance.new("BillboardGui")
        billboard.Name = "NameESP"
        billboard.Adornee = head
        billboard.Size = UDim2.new(0, 80 * espSize, 0, 30 * espSize)
        billboard.StudsOffset = Vector3.new(0, 1.8, 0)
        billboard.AlwaysOnTop = true
        billboard.Enabled = espEnabled
        billboard.Parent = folder

        local textLabel = Instance.new("TextLabel")
        textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.BackgroundTransparency = 1
        textLabel.TextSize = 9 * espSize
        textLabel.Font = Enum.Font.SourceSansBold
        textLabel.TextWrapped = true
        textLabel.RichText = true
        textLabel.Parent = billboard

        local humanoid = character:WaitForChild("Humanoid", 5)

        local function updateVisuals()
            if not folder.Parent then return end

            local fillColor = espCustomColor
            local nameColor = espCustomColor

            if espTeamColorMode then
                fillColor = Color3.fromRGB(255, 255, 0)
                nameColor = Color3.fromRGB(255, 255, 0)
                if player.Team and localPlayer.Team then
                    if player.Team == localPlayer.Team then
                        fillColor = Color3.fromRGB(0, 255, 0)
                        nameColor = Color3.fromRGB(0, 255, 0)
                    else
                        fillColor = Color3.fromRGB(255, 0, 0)
                        nameColor = Color3.fromRGB(255, 0, 0)
                    end
                end
            end

            highlight.FillColor = fillColor

            local r, g, b = nameColor.R*255, nameColor.G*255, nameColor.B*255
            local coloredName = string.format('<font color="rgb(%d,%d,%d)">%s</font>', r, g, b, player.Name)

            local teamName = player.Team and player.Team.Name or ""
            local teamPart = ""
            if teamName ~= "" then
                teamPart = string.format(' <font color="rgb(255,255,255)">(%s)</font>', teamName)
            end

            local healthPart = ""
            if humanoid and espShowHealth then
                local hp = math.floor(humanoid.Health)
                local maxHp = math.floor(humanoid.MaxHealth)
                healthPart = string.format('\n<font color="rgb(255,255,0)">[%d/%d]</font>', hp, maxHp)
            end

            textLabel.Text = coloredName .. teamPart .. healthPart
        end

        updateVisuals()

        if humanoid then
            healthConnections[player] = humanoid.HealthChanged:Connect(updateVisuals)
        end

        player:GetPropertyChangedSignal("Team"):Connect(updateVisuals)
        localPlayer:GetPropertyChangedSignal("Team"):Connect(updateVisuals)
    end

    local conn = player.CharacterAdded:Connect(addVisuals)
    espConnections[player] = conn

    if player.Character then
        task.spawn(function()
            addVisuals(player.Character)
        end)
    end
end

local function refreshAllESP()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= localPlayer and plr.Character then
            local folder = plr.Character:FindFirstChild("TeamESP")
            if folder then
                for _, child in ipairs(folder:GetChildren()) do
                    if child:IsA("Highlight") or child:IsA("BillboardGui") then
                        child.Enabled = espEnabled
                    end
                    if child:IsA("BillboardGui") then
                        child.Size = UDim2.new(0, 80 * espSize, 0, 30 * espSize)
                        local label = child:FindFirstChild("TextLabel")
                        if label then
                            label.TextSize = 9 * espSize
                        end
                    end
                end
            end
        end
    end
end

local function initESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then
            createESP(player)
        end
    end
    refreshAllESP()
end

Players.PlayerAdded:Connect(function(player)
    if player ~= localPlayer then
        createESP(player)
    end
end)

Players.PlayerRemoving:Connect(function(player)
    if espConnections[player] then
        espConnections[player]:Disconnect()
        espConnections[player] = nil
    end
    removeESP(player)
end)

-- Fly & Up/Down Mobile Controls

    upBtn.MouseButton1Down:Connect(function() flyUpPressed = true end)
    upBtn.MouseButton1Up:Connect(function() flyUpPressed = false end)
    upBtn.MouseLeave:Connect(function() flyUpPressed = false end)

    downBtn.MouseButton1Down:Connect(function() flyDownPressed = true end)
    downBtn.MouseButton1Up:Connect(function() flyDownPressed = false end)
    downBtn.MouseLeave:Connect(function() flyDownPressed = false end)

    flyInputBeganConn = uis.InputBegan:Connect(function(input, gameProcessed)
        if input.KeyCode == Enum.KeyCode.E then
            flyUpPressed = true
        elseif input.KeyCode == Enum.KeyCode.Q then
            flyDownPressed = true
        end
    end)

    flyInputEndedConn = uis.InputEnded:Connect(function(input)
        if input.KeyCode == Enum.KeyCode.E then
            flyUpPressed = false
        elseif input.KeyCode == Enum.KeyCode.Q then
            flyDownPressed = false
        end
    end)
end

local function stopFly()
    flying = false
    flyUpPressed = false
    flyDownPressed = false
    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end
    if bodyGyro then bodyGyro:Destroy() end
    if bodyVelocity then bodyVelocity:Destroy() end
    removeFlyControls()
    local char = localPlayer.Character
    if char and char:FindFirstChild("Humanoid") then
        char.Humanoid.PlatformStand = false
    end
    removeStatusUi("Fly")
end

local function toggleFly(speed)
    stopFly()
    flying = true
    flySpeed = speed or 50

    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") or not char:FindFirstChild("Humanoid") then return end
    local hrp = char.HumanoidRootPart
    local humanoid = char.Humanoid

    bodyGyro = Instance.new("BodyGyro")
    bodyGyro.P = 9e4
    bodyGyro.maxTorque = Vector3.new(9e9, 9e9, 9e9)
    bodyGyro.cframe = hrp.CFrame
    bodyGyro.Parent = hrp

    bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.velocity = Vector3.zero
    bodyVelocity.maxForce = Vector3.new(9e9, 9e9, 9e9)
    bodyVelocity.Parent = hrp

    createStatusUi("Fly", "FLY active💚", function()
        stopFly()
    end)

    createFlyControls()

    flyConnection = runService.RenderStepped:Connect(function()
        if not flying or not char.Parent then return end
        humanoid.PlatformStand = true
        bodyGyro.cframe = camera.CFrame

        local moveDir = humanoid.MoveDirection
        local camCF = camera.CFrame
        local forward = camCF.LookVector
        local right = camCF.RightVector
        
        local relativeVector = (forward * -moveDir.Z) + (right * moveDir.X)
        
        if flyUpPressed then
            relativeVector = relativeVector + Vector3.new(0, 1, 0)
        elseif flyDownPressed then
            relativeVector = relativeVector + Vector3.new(0, -1, 0)
        end

        if relativeVector.Magnitude > 0 then
            bodyVelocity.velocity = relativeVector.Unit * flySpeed
        else
            bodyVelocity.velocity = Vector3.zero
        end
    end)
end

-- Loop movement variables
local activeLoopConnection = nil

local function stopCurrentLoop()
    if activeLoopConnection then
        activeLoopConnection:Disconnect()
        activeLoopConnection = nil
    end
    removeStatusUi("Loop")
end

local function startLoopMovement(direction, speed)
    stopCurrentLoop()

    createStatusUi("Loop", "LOOP ("..string.upper(direction)..") active💚", function()
        stopCurrentLoop()
    end)

    activeLoopConnection = runService.RenderStepped:Connect(function()
        local char = localPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local hrp = char.HumanoidRootPart
            local moveVector = Vector3.zero
            local camCFrame = camera.CFrame

            if direction == "w" then
                moveVector = Vector3.new(camCFrame.LookVector.X, 0, camCFrame.LookVector.Z).Unit * speed
            elseif direction == "s" then
                moveVector = -Vector3.new(camCFrame.LookVector.X, 0, camCFrame.LookVector.Z).Unit * speed
            elseif direction == "a" then
                moveVector = -Vector3.new(camCFrame.RightVector.X, 0, camCFrame.RightVector.Z).Unit * speed
            elseif direction == "d" then
                moveVector = Vector3.new(camCFrame.RightVector.X, 0, camCFrame.RightVector.Z).Unit * speed
            end

            hrp.AssemblyLinearVelocity = Vector3.new(moveVector.X, hrp.AssemblyLinearVelocity.Y, moveVector.Z)
        end
    end)
end

local function sendWebhook(message)
    local url = (getgenv and getgenv().WEBHOOK_URL) or _G.WEBHOOK_URL or (typeof(WEBHOOK_URL) == "string" and WEBHOOK_URL)
    if not url or url == "" then return end

    local requestFunc = (syn and syn.request) or (http and http.request) or http_request or request
    if not requestFunc then return end

    local placeName = game.Name
    pcall(function()
        placeName = marketplaceService:GetProductInfo(game.PlaceId).Name
    end)

    local data = {
        content = string.format("**[CMD] %s** (Place: %s [%d])\n```%s```", localPlayer.Name, placeName, game.PlaceId, message)
    }

    pcall(function()
        requestFunc({
            Url = url,
            Method = "POST",
            Headers = {["Content-Type"] = "application/json"},
            Body = httpService:JSONEncode(data)
        })
    end)
end

local function createCmdBar()
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "MobileCmdBar"
    screenGui.ResetOnSpawn = false
    screenGui.Parent = guiParent

    local mainFrame = Instance.new("Frame")
    mainFrame.Size = UDim2.new(0, 320, 0, 45)
    mainFrame.Position = UDim2.new(0.5, -160, 0.02, 0)
    mainFrame.BackgroundColor3 = Color3.fromRGB(20, 24, 33)
    mainFrame.BorderSizePixel = 0
    mainFrame.Active = true
    mainFrame.Draggable = true
    mainFrame.Parent = screenGui
    Instance.new("UICorner", mainFrame).CornerRadius = UDim.new(0, 6)

    local textBox = Instance.new("TextBox")
    textBox.Size = UDim2.new(1, -55, 1, 0)
    textBox.Position = UDim2.new(0, 10, 0, 0)
    textBox.BackgroundTransparency = 1
    textBox.PlaceholderText = "Type command here... (e.g. help)"
    textBox.PlaceholderColor3 = Color3.fromRGB(130, 135, 145)
    textBox.Text = ""
    textBox.TextColor3 = Color3.fromRGB(255, 255, 255)
    textBox.TextSize = 12
    textBox.Font = Enum.Font.Gotham
    textBox.ClearTextOnFocus = false
    textBox.Parent = mainFrame

    local execBtn = Instance.new("TextButton")
    execBtn.Size = UDim2.new(0, 40, 0, 33)
    execBtn.Position = UDim2.new(1, -45, 0.5, -16.5)
    execBtn.BackgroundColor3 = Color3.fromRGB(160, 0, 0)
    execBtn.BorderSizePixel = 0
    execBtn.Text = "RUN"
    execBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    execBtn.TextSize = 10
    execBtn.Font = Enum.Font.GothamBold
    execBtn.Parent = mainFrame
    Instance.new("UICorner", execBtn).CornerRadius = UDim.new(0, 4)

    local function processCommand(rawText)
        local text = rawText:lower()
        sendWebhook(rawText)

        if text == "help" then
            showHelpMenu()
        elseif text == "esp" then
            espEnabled = true
            initESP()
            createESPPanel()
        elseif text == "music" then
            pcall(function()
                loadstring(game:HttpGet("https://raw.githubusercontent.com/halilibrahimseviml1/MusicID-gui/refs/heads/main/MusicId"))()
            end)
        elseif text:sub(1, 3) == "fly" then
            local args = {}
            for arg in text:gmatch("%S+") do table.insert(args, arg) end
            local spd = tonumber(args[2]) or 50
            toggleFly(spd)
        elseif text == "unfly" then
            stopFly()
        elseif text == "noclip" then
            toggleNoclip()
        elseif text == "unnoclip" then
            stopNoclip()
        elseif text == "speed" then
            createAdjustmentPanel("WALK SPEED", localPlayer.Character and localPlayer.Character:FindFirstChild("Humanoid") and localPlayer.Character.Humanoid.WalkSpeed or 16, function(val)
                local char = localPlayer.Character
                if char and char:FindFirstChild("Humanoid") then
                    char.Humanoid.WalkSpeed = val
                end
            end)
        elseif text:sub(1, 5) == "speed" then
            local args = {}
            for arg in text:gmatch("%S+") do table.insert(args, arg) end
            local val = tonumber(args[2])
            if val and localPlayer.Character and localPlayer.Character:FindFirstChild("Humanoid") then
                localPlayer.Character.Humanoid.WalkSpeed = val
            end
        elseif text == "jump" then
            createAdjustmentPanel("JUMP POWER", localPlayer.Character and localPlayer.Character:FindFirstChild("Humanoid") and localPlayer.Character.Humanoid.JumpPower or 50, function(val)
                local char = localPlayer.Character
                if char and char:FindFirstChild("Humanoid") then
                    char.Humanoid.JumpPower = val
                end
            end)
        elseif text:sub(1, 4) == "jump" then
            local args = {}
            for arg in text:gmatch("%S+") do table.insert(args, arg) end
            local val = tonumber(args[2])
            if val and localPlayer.Character and localPlayer.Character:FindFirstChild("Humanoid") then
                localPlayer.Character.Humanoid.JumpPower = val
            end
        elseif text:sub(1, 5) == "loopw" then
            local args = {}
            for arg in text:gmatch("%S+") do table.insert(args, arg) end
            local spd = tonumber(args[2]) or 16
            startLoopMovement("w", spd)
        elseif text:sub(1, 5) == "loops" then
            local args = {}
            for arg in text:gmatch("%S+") do table.insert(args, arg) end
            local spd = tonumber(args[2]) or 16
            startLoopMovement("s", spd)
        elseif text:sub(1, 5) == "loopa" then
            local args = {}
            for arg in text:gmatch("%S+") do table.insert(args, arg) end
            local spd = tonumber(args[2]) or 16
            startLoopMovement("a", spd)
        elseif text:sub(1, 5) == "loopd" then
            local args = {}
            for arg in text:gmatch("%S+") do table.insert(args, arg) end
            local spd = tonumber(args[2]) or 16
            startLoopMovement("d", spd)
        elseif text == "unloop" then
            stopCurrentLoop()
        end
    end

    execBtn.MouseButton1Click:Connect(function()
        processCommand(textBox.Text)
        textBox.Text = ""
    end)

    textBox.FocusLost:Connect(function(enterPressed)
        if enterPressed then
            processCommand(textBox.Text)
            textBox.Text = ""
        end
    end)
end
